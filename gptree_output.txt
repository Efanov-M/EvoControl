from django.shortcuts import render, get_object_or_404, redirect

from .models import Vehicle
from .forms import VehicleForm


def vehicle_list(request):
    vehicles = Vehicle.objects.all()

    return render(
        request,
        "vehicles/vehicle_list.html",
        {"vehicles": vehicles},
    )
def vehicle_detail(request, pk):
    vehicle = get_object_or_404(Vehicle, pk=pk)
    operations = vehicle.operations.all()
    return render(
        request,
        "vehicles/vehicle_detail.html",
        {
        "vehicle": vehicle,
            "operations": operations,
        },
    )

def vehicle_create(request):
    if request.method == "POST":
        form = VehicleForm(request.POST)

        if form.is_valid():
            form.save()
            return redirect("vehicle_list")

    else:
        form = VehicleForm()

    return render(
        request,
        "vehicles/vehicle_create.html",
        {
            "form": form,
        },
    )

from django.db import models

# Create your models here.
class TruckType(models.TextChoices):
    PLATFORM = "platform", "Платформа"
    MANIPULATOR = "manipulator", "Манипулятор"


class TowTruck(models.Model):
    license_plate = models.CharField(
    "Госномер",
    max_length=20,
    unique=True
    )

    brand = models.CharField(
    "Марка",
    max_length=50
    )

    truck_type = models.CharField(
    "Тип эвакуатора",
    max_length=20,
    choices=TruckType.choices,
    default=TruckType.PLATFORM
)
    model = models.CharField("Модель", max_length=20,blank=True)

    vin = models.CharField(max_length=17,blank=True, null=True, unique=True)

    is_active = models.BooleanField(default=True)

    def __str__(self):
        if self.model:
            return f"{self.brand} {self.model} ({self.license_plate})"

        return f"{self.brand} ({self.license_plate})"

from django.shortcuts import render, get_object_or_404

from .models import TowTruck


def fleet_list(request):
    tow_trucks = TowTruck.objects.all()

    return render(
        request,
        "fleet/fleet_list.html",
        {"fleet": tow_trucks},
    )
def fleet_detail(request, pk):
    tow_truck = get_object_or_404(TowTruck, pk=pk)
    operations = tow_truck.operations.all()
    return render(
        request,
        "fleet/fleet_detail.html",
        {
        "fleet": tow_truck,
        "operations": operations,
        },
    )

import os
import sys


def main():
    """Run administrative tasks."""
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'config.settings')
    try:
        from django.core.management import execute_from_command_line

    main()

# File: operations/models.py

from django.conf import settings
from django.db import models

from fleet.models import TowTruck
from vehicles.models import Vehicle


class OperationType(models.TextChoices):
    ARRIVAL = "arrival", "Приём"
    TRANSFER = "transfer", "Перемещение"
    RELEASE = "release", "Выдача"

class ParkingLot(models.Model):
    name = models.CharField("Наименование стоянки", max_length=50)
    address = models.TextField("Адрес стоянки")
    is_active = models.BooleanField(
    "Активна",
    default=True
    )

    def __str__(self):
        return self.name


class VehicleOperation(models.Model):
    vehicle = models.ForeignKey(
        Vehicle,
        on_delete=models.PROTECT,
        related_name="operations",
        verbose_name="Автомобиль",
    )

    tow_truck = models.ForeignKey(
        TowTruck,
        on_delete=models.PROTECT,
        related_name="operations",
        verbose_name="Эвакуатор",
        blank=True,
        null=True,
    )

    dispatcher = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="handled_operations",
        verbose_name="Диспетчер",
    )

    operation_type = models.CharField(
        "Тип операции",
        max_length=20,
        choices=OperationType.choices,
    )

    created_at = models.DateTimeField(
        "Дата и время операции",
        auto_now_add=True,
    )

    comment = models.TextField(
        "Комментарий",
        blank=True,
    )
    parking_lot = models.ForeignKey(
        ParkingLot,
        on_delete=models.PROTECT,
        related_name="operations",
        verbose_name="Стоянка",
        null=True,
        blank=True,
    )

    def __str__(self):
        return (
            f"{self.vehicle} — "
            f"{self.get_operation_type_display()} — "
            f"{self.created_at:%d.%m.%Y %H:%M}"
        )

from django.db import models

# Create your models here.




class Vehicle(models.Model):
    license_plate = models.CharField(
    "Госномер",
    max_length=20,
    unique=True
    )

    brand = models.CharField("Марка", max_length=50)

    model = models.CharField("Модель", max_length=20)

    color = models.CharField("Цвет", max_length=20)

    legal_basis = models.CharField("Статья изъятия", max_length=20)

    was_in_accident = models.BooleanField("Была ли в ДТП", default=False)



    def __str__(self):
        if self.was_in_accident:
            return f"{self.brand} {self.model} ({self.license_plate}) {self.color} {self.legal_basis}ДТП"

        return f"{self.brand} {self.model} ({self.license_plate}) {self.color} {self.legal_basis}"
